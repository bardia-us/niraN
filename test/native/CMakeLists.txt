cmake_minimum_required(VERSION 3.14)
project(niran_native_regressions LANGUAGES CXX)
if(NOT NIRAN_SOURCE_ROOT)
  get_filename_component(NIRAN_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_executable(process_lifecycle_test process_lifecycle_test.cpp
  "${NIRAN_SOURCE_ROOT}/windows/xray/xray_process_manager.cpp")
target_include_directories(process_lifecycle_test PRIVATE "${NIRAN_SOURCE_ROOT}")
target_compile_features(process_lifecycle_test PRIVATE cxx_std_17)
target_compile_options(process_lifecycle_test PRIVATE /W4 /WX /wd4100 /EHsc)
target_compile_definitions(process_lifecycle_test PRIVATE _HAS_EXCEPTIONS=0 UNICODE _UNICODE)
enable_testing()
add_test(NAME process_lifecycle COMMAND process_lifecycle_test)
set_tests_properties(process_lifecycle PROPERTIES TIMEOUT 20)
